export class MyKikiDemandesComponent implements OnInit {
  demandes: any[] = [];
  isLoading = true;

  constructor(private clientApi: ClientApiService, private authService: AuthService) {}

  ngOnInit(): void {
    const user = this.authService.getCurrentUser();
    if (user && user.username) {
      this.clientApi.getDemandesByEmail(user.username)
        .pipe(finalize(() => this.isLoading = false))
        .subscribe({
          next: (data) => this.demandes = data,
          error: (err) => console.error(err)
        });
    } else {
      this.isLoading = false;
    }
  }

  getStatusLabel(status: string): string {
    if (!status) return 'En cours de traitement';
    switch(status.toLowerCase()) {
      case 'pending':
      case 'en_attente': return 'En cours de traitement';
      case 'propositions_envoyees': return 'Propositions reçues';
      case 'selection_client':
      case 'proposition_validee': return 'Choix validé (en attente du devis)';
      case 'aboutis': return 'Devis validé';
      case 'rejected':
      case 'refuse': return 'Refusée / Annulée';
      default: return 'En traitement';
    }
  }

  getIconClass(status: string): string {
    if (!status) return 'fa-hourglass-half';
    switch(status.toLowerCase()) {
      case 'pending':
      case 'en_attente': return 'fa-hourglass-half';
      case 'propositions_envoyees': return 'fa-envelope-open-text';
      case 'aboutis': return 'fa-check-circle';
      case 'rejected':
      case 'refuse': return 'fa-times-circle';
      default: return 'fa-folder';
    }
  }

  getIconStyle(status: string): any {
    if (!status) return { background: 'rgba(229,29,36,0.1)', color: 'var(--color-red)' };
    switch(status.toLowerCase()) {
      case 'propositions_envoyees': return { background: 'rgba(59,130,246,0.1)', color: '#2563eb' };
      case 'aboutis': return { background: 'rgba(34,197,94,0.1)', color: '#16a34a' };
      case 'rejected':
      case 'refuse': return { background: 'rgba(220,38,38,0.1)', color: '#dc2626' };
      default: return { background: 'rgba(229,29,36,0.1)', color: 'var(--color-red)' };
    }
  }
}
